Question 1 — Food Ordering App

Create a food ordering application with a menu screen and order summary screen. Use Explicit Intents to pass order information between Activities and display a simple animation.

                }) {

                Button(onClick = {
                    val intent = Intent(
                        this@MainActivity,

    }

Question 3 — Shopping App Using Navigation Compose

Display a product list and navigate to a Product Detail screen. Pass the selected product information using Navigation Compose.

MainActivity.kt 
package com.example.shoppingnavigation

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.navigation.NavType
import androidx.navigation.compose.*
import androidx.navigation.navArgument

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            ShoppingNavigation()
        }
    }
}

@Composable
fun ShoppingNavigation() {
    val navController = rememberNavController()

    NavHost(
        navController = navController,
        startDestination = "products"
    ) {
        composable("products") {
            ProductScreen { id ->
                navController.navigate("details/$id")
            }
        }

        composable(
            route = "details/{productId}",
            arguments = listOf(
                navArgument("productId") {
                    type = NavType.IntType
                }
            )
        ) { entry ->
            val id = entry.arguments?.getInt("productId") ?: 1
            ProductDetailScreen(id) {
                navController.popBackStack()
            }
        }
    }
}

@Composable
fun ProductScreen(onProductClick: (Int) -> Unit) {
    Column(Modifier.padding(20.dp)) {
        Text("Shopping Products")

        Button(onClick = { onProductClick(1) }) {
            Text("Mobile")
        }

        Button(onClick = { onProductClick(2) }) {
            Text("Laptop")
        }

        Button(onClick = { onProductClick(3) }) {
            Text("Headphones")
        }
    }
}

@Composable
fun ProductDetailScreen(
    id: Int,
    onBack: () -> Unit
) {
    val product = when (id) {
        1 -> "Mobile - Rs.15000"
        2 -> "Laptop - Rs.50000"
        3 -> "Headphones - Rs.2000"
        else -> "Unknown Product"
    }

    Column(Modifier.padding(20.dp)) {
        Text(
            "Product Details",
            style = MaterialTheme.typography.headlineMedium
        )

        Text(product)

        Button(onClick = onBack) {
            Text("Back")
        }
    }
}

Question 5 — Background Sync using WorkManager

Create a background synchronization application with Start Sync and Cancel Sync buttons. Schedule periodic work every 15 minutes, log the current time, and display the WorkManager status.

SyncWorker.kt 
package com.example.backgroundsync

import android.content.Context
import android.util.Log
import androidx.work.Worker
import androidx.work.WorkerParameters
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

class SyncWorker(
    context: Context,
    params: WorkerParameters
) : Worker(context, params) {

    override fun doWork(): Result {

        val currentTime = SimpleDateFormat(
            "HH:mm:ss",
            Locale.getDefault()
        ).format(Date())

        Log.d("SyncWorker", "Sync performed at $currentTime")

        return Result.success()
    }
}
MainActivity.kt 
package com.example.backgroundsync

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.work.*
import kotlinx.coroutines.flow.collectLatest
import java.util.concurrent.TimeUnit

class MainActivity : ComponentActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        setContent {

            var status by remember {
                mutableStateOf("Not Started")
            }

            val workManager = WorkManager.getInstance(this)

            LaunchedEffect(Unit) {
                workManager
                    .getWorkInfosForUniqueWorkFlow("periodic_sync")
                    .collectLatest { workInfos ->

                        status = if (workInfos.isEmpty()) {
                            "Not Started"
                        } else {
                            workInfos.first().state.name
                        }
                    }
            }

            Column(
                modifier = Modifier.padding(24.dp)
            ) {

                Text(
                    "Background Sync",
                    style = MaterialTheme.typography.headlineMedium
                )

                Text("Status: $status")

                Button(onClick = {

                    val request =
                        PeriodicWorkRequestBuilder(
                            15, TimeUnit.MINUTES
                        ).build()

                    workManager.enqueueUniquePeriodicWork(
                        "periodic_sync",
                        ExistingPeriodicWorkPolicy.KEEP,
                        request
                    )

                }) {
                    Text("Start Sync")
                }

                Button(onClick = {

                    workManager.cancelUniqueWork("periodic_sync")

                }) {
                    Text("Cancel Sync")
                }
            }
        }
    }
}
build.gradle.kts — Dependency 
dependencies {
    implementation("androidx.work:work-runtime-ktx:2.10.5")
}
Remember:
Worker performs background work.
doWork() contains the task.
PeriodicWorkRequestBuilder() schedules repeated work.
15 minutes is the minimum periodic interval.
enqueueUniquePeriodicWork() schedules unique work.
cancelUniqueWork() cancels it.
WorkInfo.state gives ENQUEUED, RUNNING, SUCCEEDED, or CANCELLED.

Exam note: Periodic work normally remains ENQUEUED between runs. The worker's Result.success() does not make periodic work permanently SUCCEEDED; it schedules another execution.

BookDao.kt — Database Operations 
package com.example.readinglist

import androidx.room.*
import kotlinx.coroutines.flow.Flow

@Dao
interface BookDao {

    @Insert
    suspend fun insertBook(book: Book)

    @Delete
    suspend fun deleteBook(book: Book)

    @Query("SELECT * FROM books ORDER BY title ASC")
    fun getBooksByTitle(): Flow<List<Book>>

    @Query("SELECT * FROM books ORDER BY timestamp DESC")
    fun getBooksNewestFirst(): Flow<List<Book>>
}
BookDatabase.kt — Room Database 
package com.example.readinglist

import android.content.Context
import androidx.room.Database
import androidx.room.Room
import androidx.room.RoomDatabase

@Database(
    entities = [Book::class],
    version = 1
)
abstract class BookDatabase : RoomDatabase() {

    abstract fun bookDao(): BookDao

    companion object {

        @Volatile
        private var INSTANCE: BookDatabase? = null

        fun getDatabase(context: Context): BookDatabase {

            return INSTANCE ?: synchronized(this) {

                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    BookDatabase::class.java,
                    "reading_database"
                ).build()

                INSTANCE = instance
                instance
            }
        }
    }
}
BookPreferences.kt — DataStore 
package com.example.readinglist

import android.content.Context
import androidx.datastore.preferences.core.edit
import androidx.datastore.preferences.core.stringPreferencesKey
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map

private val Context.dataStore by preferencesDataStore(
    name = "book_settings"
)

class BookPreferences(private val context: Context) {

    companion object {
        val SORT_ORDER = stringPreferencesKey("sort_order")
    }

    val sortOrder: Flow<String> =
        context.dataStore.data.map { preferences ->
            preferences[SORT_ORDER] ?: "TITLE"
        }

    suspend fun saveSortOrder(order: String) {
        context.dataStore.edit { preferences ->
            preferences[SORT_ORDER] = order
        }
    }
}
BookViewModel.kt — Logic 
package com.example.readinglist

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class BookViewModel(
    application: Application
) : AndroidViewModel(application) {

    private val dao = BookDatabase
        .getDatabase(application)
        .bookDao()

    private val preferences = BookPreferences(application)

    private val sortOrder = preferences.sortOrder

    val books: StateFlow<List<Book>> =
        sortOrder.flatMapLatest { order ->

            if (order == "NEWEST") {
                dao.getBooksNewestFirst()
            } else {
                dao.getBooksByTitle()
            }

        }.stateIn(
            viewModelScope,
            SharingStarted.WhileSubscribed(5000),
            emptyList()
        )

    fun addBook(title: String, author: String) {
        viewModelScope.launch {
            dao.insertBook(
                Book(title = title, author = author)
            )
        }
    }

    fun deleteBook(book: Book) {
        viewModelScope.launch {
            dao.deleteBook(book)
        }
    }

    fun changeSort(order: String) {
        viewModelScope.launch {
            preferences.saveSortOrder(order)
        }
    }
}

package com.example.commentsapp

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import kotlinx.coroutines.delay

data class Comment(
    val id: Int,
    val name: String,
    val email: String
)

// Local data source for offline demonstration
object CommentRepository {
    fun getComments(): List<Comment> {
        return listOf(
            Comment(1, "John", "john@gmail.com"),
            Comment(2, "David", "david@gmail.com"),
            Comment(3, "Mary", "mary@gmail.com"),
            Comment(4, "Alex", "alex@gmail.com"),
            Comment(5, "Peter", "peter@gmail.com")
        )
    }
}

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            CommentsScreen()
        }
    }
}

@Composable
fun CommentsScreen() {
    var comments by remember {
        mutableStateOf(emptyList<Comment>())
    }

    var isLoading by remember {
        mutableStateOf(true)
    }

    // Simulate API loading
    LaunchedEffect(Unit) {
        delay(1500)
        comments = CommentRepository.getComments()
        isLoading = false
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp)
    ) {
        Text(
            text = "Comments List",
            style = MaterialTheme.typography.headlineMedium
        )

        Spacer(modifier = Modifier.height(16.dp))

        if (isLoading) {
            CircularProgressIndicator()
            Text("Loading comments...")
        } else {
            LazyColumn {
                items(comments) { comment ->
                    Card(
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(vertical = 6.dp)
                    ) {
                        Column(
                            modifier = Modifier.padding(16.dp)
                        ) {
                            Text(
                                text = comment.name,
                                style = MaterialTheme.typography.titleMedium
                            )

                            Text(text = comment.email)
                        }
                    }
                }
            }
        }
    }
}

Important points to remember

  • Comment is the data class.
  • CommentRepository supplies sample data.
  • LaunchedEffect runs the loading operation.
  • delay() simulates network waiting time.
  • CircularProgressIndicator() displays the loading symbol.
  • LazyColumn displays the comments in a scrollable list.

Actual API URL in the question: https://jsonplaceholder.typicode.com/comments

Result: Comments are displayed with their names and email addresses after a simulated loading period.

Experiment 9 — Feedback Collector (Offline Firestore Simulation)

Aim: Accept a user's name and comment, submit feedback, and display submitted feedback in a scrollable list.

Offline note: This version uses an in-memory list to simulate the real-time Firestore screen. It does not connect to Firebase or permanently store data.

MainActivity.kt

package com.example.feedbackapp

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

data class Feedback(
    val name: String,
    val comment: String,
    val timestamp: String
)

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            FeedbackScreen()
        }
    }
}

@Composable
fun FeedbackScreen() {
    var name by remember { mutableStateOf("") }
    var comment by remember { mutableStateOf("") }

    // In-memory list simulates Firestore updates
    val feedbackList = remember {
        mutableStateListOf(
            Feedback("John", "Good application!", "Sample"),
            Feedback("Mary", "Very useful.", "Sample")
        )
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp)
    ) {
        Text(
            text = "Feedback Collector",
            style = MaterialTheme.typography.headlineMedium
        )

        Spacer(modifier = Modifier.height(12.dp))

        OutlinedTextField(
            value = name,
            onValueChange = { name = it },
            label = { Text("Enter Name") },
            modifier = Modifier.fillMaxWidth()
        )

        Spacer(modifier = Modifier.height(8.dp))

        OutlinedTextField(
            value = comment,
            onValueChange = { comment = it },
            label = { Text("Enter Comment") },
            modifier = Modifier.fillMaxWidth()
        )

        Spacer(modifier = Modifier.height(12.dp))

        Button(
            onClick = {
                if (name.isNotBlank() && comment.isNotBlank()) {

                    val currentTime =
                        SimpleDateFormat(
                            "dd-MM-yyyy HH:mm",
                            Locale.getDefault()
                        ).format(Date())

                    feedbackList.add(
                        Feedback(
                            name = name,
                            comment = comment,
                            timestamp = currentTime
                        )
                    )

                    name = ""
                    comment = ""
                }
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text("Submit Feedback")
        }

        Spacer(modifier = Modifier.height(16.dp))

        Text(
            text = "Submitted Feedback",
            style = MaterialTheme.typography.titleLarge
        )

        LazyColumn {
            items(feedbackList) { feedback ->
                Card(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(vertical = 5.dp)
                ) {
                    Column(
                        modifier = Modifier.padding(12.dp)
                    ) {
                        Text(
                            text = feedback.name,
                            style = MaterialTheme.typography.titleMedium
                        )

                        Text(text = feedback.comment)

                        Text(
                            text = feedback.timestamp,
                            style = MaterialTheme.typography.bodySmall
                        )
                    }
                }
            }
        }
    }
}

Important points to remember

  • OutlinedTextField accepts the name and comment.
  • mutableStateListOf() stores feedback and updates the UI when an item is added.
  • Button submits the feedback.
  • isNotBlank() prevents empty submissions.
  • SimpleDateFormat creates a display timestamp.
  • LazyColumn displays feedback in a scrollable list.

Real Firestore concept: In the actual Firebase version, feedback documents would be added to the feedback collection with a server timestamp, and a snapshot listener would update the list in real time.

Result: Users can submit feedback and immediately see it added to the list.

Experiment 10 — Tip Calculator with Unit and UI Testing

Aim: Create a tip calculator that accepts a bill amount, allows selection of 10%, 15%, or 20% tip, and displays the tip and total amount.

1. TipCalculator.kt — Calculation Logic

package com.example.tipcalculator

class TipCalculator {

    fun calculateTip(
        amount: Double,
        tipPercent: Int
    ): Double {
        return amount * tipPercent / 100.0
    }

    fun calculateTotal(
        amount: Double,
        tipPercent: Int
    ): Double {
        return amount + calculateTip(amount, tipPercent)
    }
}

2. MainActivity.kt — User Interface

package com.example.tipcalculator

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import java.util.Locale

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            TipScreen()
        }
    }
}

@Composable
fun TipScreen() {
    var amount by remember { mutableStateOf("") }
    var tipPercent by remember { mutableIntStateOf(10) }

    var tip by remember { mutableDoubleStateOf(0.0) }
    var total by remember { mutableDoubleStateOf(0.0) }

    val calculator = remember { TipCalculator() }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(20.dp)
    ) {
        Text(
            text = "Tip Calculator",
            style = MaterialTheme.typography.headlineMedium
        )

        Spacer(modifier = Modifier.height(16.dp))

        OutlinedTextField(
            value = amount,
            onValueChange = { amount = it },
            label = { Text("Bill Amount") },
            modifier = Modifier.fillMaxWidth()
        )

        Spacer(modifier = Modifier.height(16.dp))

        Text("Select Tip Percentage")

        Row {
            listOf(10, 15, 20).forEach { percent ->
                Row {
                    RadioButton(
                        selected = tipPercent == percent,
                        onClick = { tipPercent = percent }
                    )
                    Text(
                        text = "$percent%",
                        modifier = Modifier.padding(top = 12.dp)
                    )
                }
            }
        }

        Spacer(modifier = Modifier.height(12.dp))

        Button(
            onClick = {
                val bill = amount.toDoubleOrNull() ?: 0.0

                tip = calculator.calculateTip(
                    bill,
                    tipPercent
                )

                total = calculator.calculateTotal(
                    bill,
                    tipPercent
                )
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text("Calculate")
        }

        Spacer(modifier = Modifier.height(20.dp))

        Text(
            text = "Tip: Rs.${String.format(Locale.US, "%.2f", tip)}"
        )

        Text(
            text = "Total: Rs.${String.format(Locale.US, "%.2f", total)}"
        )
    }
}

5. Deliberately Introduce a Bug

To demonstrate a failing test, temporarily change the calculation line in TipCalculator.kt:

// WRONG CODE — deliberate bug
return amount * tipPercent / 10.0

Expected failure: For a bill of Rs.100 and a 10% tip, the buggy code calculates Rs.100 instead of Rs.10. The unit test fails.

6. Fix the Bug

Restore the correct calculation:

// CORRECT CODE
return amount * tipPercent / 100.0

Run the tests again. The expected results are:

  • calculateTip_isCorrect() — Passed
  • calculateTotal_isCorrect() — Passed
  • calculateFifteenPercentTip_isCorrect() — Passed
  • tipCalculator_displaysCorrectResult() — Passed
